cmake_minimum_required(VERSION 3.16)
project(sbp_safety_gate C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
add_library(sbp_safety_gate safety_gate.c)
target_include_directories(sbp_safety_gate PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
add_executable(safety_gate_test test_safety_gate.c)
target_link_libraries(safety_gate_test PRIVATE sbp_safety_gate)
enable_testing()
add_test(NAME safety_gate_test COMMAND safety_gate_test)
